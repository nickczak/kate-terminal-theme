cmake_minimum_required(VERSION 3.20)
project(kate_terminal_theme VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6TextEditor REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Parts REQUIRED)

include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES Qt6::Widgets KF6::Parts)
check_cxx_source_compiles("#include <kde_terminal_interface.h>
int main() { using ProfileSetter = decltype(&TerminalInterface::setCurrentProfile); return 0; }"
    HAS_TERMINAL_PROFILE_INTERFACE)
if(NOT HAS_TERMINAL_PROFILE_INTERFACE)
    message(FATAL_ERROR "KF6 Parts lacks TerminalInterface::setCurrentProfile. Update KDE Frameworks before installing this setup.")
endif()
unset(CMAKE_REQUIRED_LIBRARIES)

add_library(kateterminalex MODULE plugin.cpp)
set_target_properties(kateterminalex PROPERTIES PREFIX "")
target_link_libraries(kateterminalex PRIVATE KF6::TextEditor KF6::CoreAddons KF6::Parts Qt6::Widgets)

option(BUILD_TESTING "Build the plugin behavior test" OFF)
option(BUILD_KATE_INTEGRATION_TESTS "Build the test-only Kate probe (never install it)" OFF)
option(BUILD_SHOWCASE_RECORDER "Build the development-only native UI recorder" OFF)
if(BUILD_TESTING OR BUILD_KATE_INTEGRATION_TESTS OR BUILD_SHOWCASE_RECORDER)
    find_package(Qt6 REQUIRED COMPONENTS Test)
endif()
if(BUILD_TESTING)
    enable_testing()
    add_executable(test-ex test.cpp)
    target_link_libraries(test-ex PRIVATE KF6::TextEditor KF6::CoreAddons KF6::Parts Qt6::Widgets Qt6::Test)
    add_test(NAME explorer-behavior COMMAND test-ex $<TARGET_FILE:kateterminalex>)
    set_tests_properties(explorer-behavior PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(BUILD_KATE_INTEGRATION_TESTS)
    add_library(exprobe MODULE integration_probe.cpp)
    # The probe intentionally uses its own ID and is only loaded in a temporary test session.
    target_link_libraries(exprobe PRIVATE KF6::TextEditor KF6::CoreAddons KF6::Parts Qt6::Widgets Qt6::Test)
endif()

if(BUILD_SHOWCASE_RECORDER)
    add_library(showcase_recorder MODULE showcase_recorder.cpp)
    set_target_properties(showcase_recorder PROPERTIES PREFIX "")
    target_link_libraries(showcase_recorder PRIVATE KF6::TextEditor KF6::CoreAddons KF6::Parts Qt6::Widgets Qt6::Test)
endif()
